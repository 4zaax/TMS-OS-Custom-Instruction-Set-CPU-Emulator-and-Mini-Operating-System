_MAIN:
    showstr [0]     # show startup message
    _MAIN_LOOP:
    showstr [80]
    AS idx 500      # user enters commands from memory house  500
    call get_command
    call do_command
    JMP _MAIN_LOOP
END_MAIN:

get_command:
    push idx
    call get_line
    pop idx
    ret
END_get_command:


get_line:
    get c
    EQ eax c '\n'
    MOV [idx] '\0'
    RET eax
    MOV [idx] c
    INC idx
    JMP get_line
end_get_line:

save_char:
    MOV [idx] c
    INC idx
    RET
end_save_char:


do_command:
        call find_command
        je END eax 1
        je command_error eax 0
        command_error:
            showstr [60]
        RET
end_do_command:


######################## turns eax 1 if command is equal to shut down ########################################################################
find_command:
    MOV ecx idx   # start index of user command
    MOV edx 90    # start index of shut down mssg command
    call str_cmp
    as eax efx
    RET
end_find_command:

############################# str_cmp checks ecx and edx turns efx 1 if 2 strings are equal and 0 if not ################################################
str_cmp:
    push ecx
    push edx
    AS efx 1
    str_cmp_loop:
        push ecx
        push edx
        mov ecx [ecx]
        mov edx [edx]
        eq efx ecx edx
        je finish efx 0 # characters were not equal so efx is 0
        eq efx ecx 0
        jmp finish efx  # characters were equal and both were 0 so efx should be 1
        ########################################################################################################################
        # now if none of those conditions are true it means that we have equal characters and string has not reached the end yet
        ########################################################################################################################
        pop edx
        pop ecx
        inc ecx
        inc edx
        mov ecx [ecx]
        mov edx [edx]
        jmp str_cmp_loop
        finish:
            pop edx
            pop ecx
        end_finish:
    end_str_cmp_loop:
    pop edx
    pop ecx
    ret
end_str_cmp:


END:

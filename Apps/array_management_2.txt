_array_management_start:

	mov v25 v21

_array_management_main_loop:
    mov esi msg_directory
	movb ah 0x09
	int 21h
	mov esi hp
	call _getstr
	mov v1 esi
	call _check_user_command
	jmp _array_management_main_loop

_getstr:
    # receives one character
    movb ah 0x08
    int 21h
    eq bl al 10
    jmp _getstr_end bl
    mov [hp] al
    inc hp
    mov [hp] 10
    jmp _getstr
_getstr_end:
    mov [hp] 10
    inc hp
    mov [hp] 0
    inc hp
    ret

_check_user_command:

    call ar_check_for_exit
    jmp _exit_handler bl

	jmp _check_user_command_end
_check_user_command_end:
	ret

_help_handler:
	mov esi msg_help
	movb ah 0x09
	int 21h

	RET

_add_handler:

    call _get_number_count
    mov v25 v21
    mov v20 0
    call _get_array_numbers
    ret

_inorder_handler:
    #bubble sort but always consider it in worst case
    mov edx v29
    ;mov v25 v21
    mov v10 1
    mov v26 v21
    mov v15 0
    call _sort
    mov v20 0
    mov v26 v21
    call _print_array
    ret

_max_handler:
    #set max v9=max
    mov v9 0
    mov v26 v21
    mov v20 0
    call _find_max
    mov esi msg_max
    movb ah 0x09
    int 21h
    mov v2 v9
    call _print_int
    ret

_min_handler:
    #set max v9=min limit 1mil
    mov v9 1000000
    mov v26 v21
    mov v20 0
    call _find_min
    mov esi msg_min
    movb ah 0x09
    int 21h
    mov v2 v9
    call _print_int
    ret



_deorder_handler:
    #bubble sort but always consider it in worst case
    ;mov v25 v21
    mov edx v29
    mov v10 1
    mov v26 v21
    mov v15 0
    call _sort
    mov v20 0
    mov v7 v8
    sub v7 v7 4
    mov v26 v7
    call _print_dec_array
    ret

_history_handler:
    push esi
    mov esi msg_history
    movb ah 0x09
    int 21h

    push v1
    pushb bh
    pushb bl
    push edx
    mov v1 v0

    _history_handler_main_loop:
        eq bh v1 hp
        jmp _history_handler_end bh

        movb dl [v1]
        neq bl dl 0
        jmp _history_handler_putc bl
        inc v1
        jmp _history_handler_main_loop

    _history_handler_putc:
        movb ah 0x02
        int 21h
        inc v1
        jmp _history_handler_main_loop

_history_handler_end:
    pop edx
    popb bl
    popb bh
    pop v1
    pop esi
    ret

_clh_handler:
    mov hp v0
    ret

_avg_handler:
    #set avg v9=avg
    mov v9 0
    ; iterator is 26
    ; v20 counts until edx
    ; 21 points to first
    ; 25 points to last
    ; edx added number count

    mov v26 v21
    mov v20 0


    call _find_sum


    mov v19 edx
    div v9 v9 v19
    mov esi msg_div
    movb ah 0x09
    int 21h
    mov v2 v9
    call _print_int
    ret

_exit_handler:
    hlt

_get_number_count:
    mov esi msg_add
    movb ah 0x09
    int 21h

    call _read_int
    mov edx v1
    mov v29 edx
    jmp _get_number_count_end
_get_number_count_end:
    ret

_get_array_numbers:

    #v25 is a pointer to the first empty array indx
    #v21 is the first indx of array
    #v20 for my number count

    #while condition
    eq bl v20 edx
    jmp _get_array_numbers_end bl

    mov esi msg_add2
    movb ah 0x09
    int 21h

    call _read_int
    mov [v25] v1
    add v25 v25 4
    inc v20

    jmp _get_array_numbers
_get_array_numbers_end:
    #return to the first array indx
    #v10 is a pointer for sort
    mov v10 1
    mov v8 v25
    ret

_sort:

    eq bl v10 edx
    jmp _sort_end bl
    mov v17 [v26]
    add v26 v26 4
    mov ecx [v26]
    inc v10
    call _comp_two_numbers
    jmp _sort
_sort_end:

    inc v15
    neq bl v15 edx
    jmp _one_more_sort bl

    ret

_one_more_sort:
    mov v26 v21
    mov v10 1
    jmp _sort

_comp_two_numbers:
    gte bl v17 ecx
    call _first_is_greater bl
    ret

_first_is_greater:
    mov v27 v26
    mov [v27] v17
    sub v27 v27 4
    mov [v27] ecx
    ret
_print_array:
    eq bl v20 edx
    jmp _print_array_end bl
    mov v2 [v26]
    call _print_int
    add v26 v26 4
    inc v20
    jmp _print_array
_print_array_end:
    ret

_print_dec_array:
    eq bl v20 edx
    jmp _print_array_dec_end bl
    mov v2 [v26]
    call _print_int
    sub v26 v26 4
    inc v20
    jmp _print_dec_array
_print_array_dec_end:
    ret

_find_max:
    eq bl v20 edx
    jmp _find_max_end bl
    gte bl [v26] v9
    call _is_bigger bl
    add v26 v26 4
    inc v20
    jmp _find_max
_find_max_end:
    ret

_is_bigger:
    mov v9 [v26]
    ret

_find_min:
    eq bl v20 edx
    jmp _find_min_end bl
    lse bl [v26] v9
    call _is_smaller bl
    add v26 v26 4
    inc v20
    jmp _find_min
_find_min_end:
    ret

_is_smaller:
    mov v9 [v26]
    ret

_find_sum:
    eq bl v20 edx
    jmp _find_min_end bl
    add v9 v9 [v26]
    add v26 v26 4
    inc v20
    jmp _find_sum
_find_sum_end:
    ret


###########################################read int
_read_int:
    mov v1 0
_read_int_loop:
    ; first receive one character put it in al
    movb ah 0x08
    int 21h

    eq bl al 32
    #for space

    eq cl al 10
    #for enter

    orb bl bl cl
    jmp _read_int_end bl
    jmp is_digit 1
is_digit:
    subb bl al '0'
    IMUL v1 v1 10
    ADD v1 v1 bl
    jmp _read_int_loop 1
_read_int_end:
    RET
###########################################read int


###########################################print int
_print_int:
    push ecx
    push edx
    push v3
    push v2
    jmp _print_loop:
_print_loop:
    eq bl v2 0
    jmp _print_end_split bl
    div v2 v2 10
    add edx edx '0'
    mov [hp] edx
    inc hp
    inc v3
    jmp _print_loop
_print_end_split:
    mov v4 0
    dec hp
    jmp _print_loop_2
_print_loop_2:
    eq bl v3 v4
    jmp _print_end_put bl
    movb dl [hp]
    movb ah 0x02
    int 21h
    inc v4
    dec hp
    jmp _print_loop_2
_print_end_put:
    movb dl 10
    movb ah 0x02
    int 21h
    pop v2
    pop v3
    pop edx
    pop ecx
    ret
#######################################print int

_initial:
MOV cs 0
MOV ds 10000
MOV ss 13000
MOV esp 19998

_terminal_start:
    msg_directory: db "~/" 0
    msg_help: db "help:     -helps you right? brilliant" 10 "echo:          -shows message on screen" 10 0
    msg2: db "echo encountered." 10 0
    msg3: db "this is our new mini os" 10 0
    msg_history: db "Showing commands history: (type clh to clear ram and free up memory)" 10 0
    name1: db "shayan" 0
    name2: db "tannaz" 0
    name3: db "mani" 0
    name4: db "shahab" 0
    name5: db "shayan" 0
	show hp
	mov v0 hp
_terminal_main_loop:
	
	# prints directory first
	mov esi msg_directory
	movb ah 0x09
	int 21h
	mov esi hp
	;delete these lines :
	;mov esi name1
	;mov ecx name5
	;call _strcmp
	;showb al
	call _getstr

	movb ah 0x09
	int 21h
	mov v1 esi
	call _check_user_command
	jmp _terminal_main_loop

_getstr:
    # receives one character
    movb ah 0x08
    int 21h
    eq bl al 10
    jmp _getstr_end bl
    mov [hp] al
    inc hp
    mov [hp] 10
    jmp _getstr
_getstr_end:
    mov [hp] 10
    inc hp
    mov [hp] 0
    inc hp
    ret

_check_user_command:

	# each check function sets bl 1 if the check result is true
	call _check_for_echo
	call _echo_handler bl		
	
	call _check_for_help
	call _help_handler bl

	call _check_for_halt
	call _halt_handler bl

	call _check_for_history
	call _history_handler bl

	call _check_for_clh
	call _clh_handler bl


    call _check_for_strcmp
    call _strcmp bl


	
	JMP _check_user_command_end
	
	_check_for_echo:
		mov v2 v1
		
		MOVb bh [v2]
		neq bl bh 'e'
		jmp _check_for_echo_end bl
		inc v2
		
		MOVB bh [v2]
		neq bl bh 'c'
		jmp _check_for_echo_end bl
		inc v2		
		
		MOVB bh [v2]
		neq bl bh 'h'
		jmp _check_for_echo_end bl
		inc v2		
		
		MOVB bh [v2]
		neq bl bh 'o'
		jmp _check_for_echo_end bl
		inc v2
		
		MOVB bl 1
		MOV v2 0
		ret
		
		_check_for_echo_end:
		MOVB bl 0
		ret	
		
	_check_for_help:
		mov v2 v1
		
		MOVb bh [v2]
		neq bl bh 'h'
		jmp _check_for_help_end bl
		inc v2
		
		MOVB bh [v2]
		neq bl bh 'e'
		jmp _check_for_help_end bl
		inc v2		
		
		MOVB bh [v2]
		neq bl bh 'l'
		jmp _check_for_help_end bl
		inc v2		
		
		MOVB bh [v2]
		neq bl bh 'p'
		jmp _check_for_help_end bl
		inc v2
		
		MOVB bl 1
		MOV v2 0
		ret
		
		_check_for_help_end:
		MOVB bl 0
		ret


		_check_for_halt:
		    mov v2 v1
            MOVb bh [v2]
            neq bl bh 'h'
            jmp _check_for_help_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 'l'
            jmp _check_for_help_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 't'
            jmp _check_for_help_end bl
            inc v2

            MOVB bl 1
            MOV v2 0
            ret

        _check_for_history:
            mov v2 v1

            MOVb bh [v2]
            neq bl bh 'h'
            jmp _check_for_history_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 'i'
            jmp _check_for_history_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 's'
            jmp _check_for_history_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 't'
            jmp _check_for_history_end bl
            inc v2

            MOVB bl 1
            MOV v2 0
            ret

            _check_for_history_end:
            MOVB bl 0
            ret


        _check_for_clh:
            mov v2 v1

            MOVb bh [v2]
            neq bl bh 'c'
            jmp _check_for_clh_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 'l'
            jmp _check_for_clh_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 'h'
            jmp _check_for_clh_end bl
            inc v2


            MOVB bl 1
            MOV v2 0
            ret

            _check_for_clh_end:
            MOVB bl 0
            ret

        _check_for_strcmp:
            mov v2 v1

            MOVb bh [v2]
            neq bl bh 's'
            jmp _check_for_strcmp_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 't'
            jmp _check_for_strcmp_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 'r'
            jmp _check_for_strcmp_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 'c'
            jmp _check_for_strcmp_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 'm'
            jmp _check_for_strcmp_end bl
            inc v2

            MOVB bh [v2]
            neq bl bh 'p'
            jmp _check_for_strcmp_end bl
            inc v2


            MOVB bl 1
            MOV v2 0
            ret

            _check_for_strcmp_end:
            MOVB bl 0
            ret
		
		
	_check_user_command_end:
	RET
	
	
_echo_handler:
	mov esi msg2
	movb ah 0x09
	int 21h
	
	RET

_help_handler:
	mov esi msg_help
	movb ah 0x09
	int 21h
	
	RET


_halt_handler:
    hlt

_history_handler:
    push esi
    mov esi msg_history
    movb ah 0x09
    int 21h

    push v1
    pushb bh
    pushb bl
    mov v1 v0

    _history_handler_main_loop:
        eq bh v1 hp
        jmp _history_handler_end bh

        movb dl [v1]
        neq bl dl 0
        jmp _history_handler_putc bl
        inc v1
        jmp _history_handler_main_loop

    _history_handler_putc:
        movb ah 0x02
        int 21h
        inc v1
        jmp _history_handler_main_loop

    _history_handler_end:
        popb bl
        popb bh
        pop v1
        pop esi
        ret

_clh_handler:
    mov hp v0
    ret

# expect first string to be in esi register and the second string pointer to be in ecx -> sets al to 0 if those 2 are not equal
_strcmp:
    MOVB al 0
    push esi
    push ecx
    mov esi name5
    _strcmp_main:
        movb ah [esi]
        movb bh [ecx]
        neq al ah bh
        jmp _strcmp_end al
        eq bl ah 0
        jmp _strcmp_eq bl
        inc esi
        inc ecx
        jmp _strcmp_main

_strcmp_eq:
    movb al 1
    pop ecx
    pop esi
    ret
_strcmp_end:
    pop ecx
    pop esi
    ret

# put your string in esi call strlen and receive length in eax register
_strlen:
    push esi
    pushb bl
    mov eax 0
    _strlen_main:
        eq  bl [esi] 0
        jmp _strlen_end bl
        inc eax
        inc esi
        jmp _strlen_main
_strlen_end:
    popb bl
    pop esi
    ret

_initial:
MOV cs 0
MOV ds 10000
MOV ss 13000
MOV esp 20000

################################################
jmp start

INPUT:
    movb ah 0x08
    int 21h
    mov [hp] al
    neq bl al 10
    inc hp
    jmp INPUT bl
    ret


start:
call INPUT
###########################################
# hp is 10058

call _getstr
mov esi 10058
movb ah 0x09
int 21h
hlt

_getstr:
    # receives one character
    movb ah 0x08
    int 21h
    eq bl al 10
    jmp _getstr_end bl
    mov [hp] al
    inc hp
    mov [hp] 10
    jmp _getstr
_getstr_end:
    mov [hp] 10
    inc hp
    mov [hp] 0
    inc hp
    ret

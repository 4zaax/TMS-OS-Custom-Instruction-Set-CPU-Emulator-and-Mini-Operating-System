_getstr:
    # receives one character
    movb ah 0x08
    int 21h
    eq bl al 10
    jmp _getstr_end bl
    mov [hp] al
    inc hp
    mov [hp] 10
    jmp _getstr
_getstr_end:
    mov [hp] 10
    inc hp
    mov [hp] 0
    inc hp

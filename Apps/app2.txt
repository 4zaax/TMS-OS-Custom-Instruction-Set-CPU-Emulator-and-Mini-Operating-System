_initial:
MOV cs 0
MOV ds 10000
MOV ss 13000
MOV esp 19998


msg: db "shayan" 0

_start:
    mov esi msg
    show esi
    showb bl
    call _strlen
    show eax
    show esi
    showb bl
    jmp _start

# put your string in esi call strlen and receive length in eax register
_strlen:
    push esi
    pushb bl
    mov eax 0
    _strlen_main:
        eq  bl [esi] 0
        jmp _strlen_end bl
        inc eax
        inc esi
        jmp _strlen_main
_strlen_end:
    popb bl
    pop esi
    ret

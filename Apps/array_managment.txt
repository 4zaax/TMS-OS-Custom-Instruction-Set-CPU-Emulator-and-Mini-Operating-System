_history_handler:
    push esi
    mov esi msg_history
    movb ah 0x09
    int 21h

    push v1
    pushb bh
    pushb bl
    mov v1 v0

    _history_handler_main_loop:
        eq bh v1 hp
        jmp _history_handler_end bh

        movb dl [v1]
        neq bl dl 0
        jmp _history_handler_putc bl
        inc v1
        jmp _history_handler_main_loop

    _history_handler_putc:
        movb ah 0x02
        int 21h
        inc v1
        jmp _history_handler_main_loop

_history_handler_end:
    popb bl
    popb bh
    pop v1
    pop esi
    ret

_clh_handler:
    mov hp v0
    ret

_get_number_count:
    mov esi msg_add
    movb ah 0x09
    int 21h

    call _read_int
    mov edx v1
    jmp _get_number_count_end
_get_number_count_end:
    show edx
    ret

_get_array_numbers:


    #v25 is a pointer to the first empty array indx
    #v21 is the first indx of array
    #v20 for my number count

    #while condition
    eq bl v20 edx
    jmp _get_array_numbers_end bl

    mov esi msg_add2
    movb ah 0x09
    int 21h

    call _read_int
    mov [v25] v1
    add v25 v25 4
    inc v20

    jmp _get_array_numbers
_get_array_numbers_end:
    #return to the first array indx
    #v10 is a pointer for sort
    mov v10 1
    ret

_sort:

    eq bl v10 edx
    jmp _sort_end bl
    mov v17 [v26]
    add v26 v26 4
    mov ecx [v26]
    show v17
    show ecx
    inc v10
    call _comp_two_numbers
    mov esi msg_bebbug
    int 21h
    show v26
    showd_mem v26
    jmp _sort
_sort_end:

    inc v15
    show v15
    neq bl v15 edx
    jmp _one_more_sort bl

    ret

_one_more_sort:
    mov v26 v21
    mov v10 1
    jmp _sort

_comp_two_numbers:
    debugger
    gte bl v17 ecx
    showb bl
    call _first_is_greater bl
    ret

_first_is_greater:
    mov v27 v26
    showd_mem v27
    show v17
    mov [v27] v17
    showd_mem v27
    sub v27 v27 4
    showd_mem v27
    mov [v27] ecx
    showd_mem v27
    ret
_print_array:
    eq bl v20 edx
    jmp _print_array_end bl
    showd_mem v26
    add v26 v26 4
    inc v20
    jmp _print_array
_print_array_end:
    ret

_find_max:
    eq bl v20 edx
    jmp _find_max_end bl
    gte bl [v25] v10
    call _is_bigger
    add v25 v25 4
    inc v20
    jmp _find_max
_find_max_end:
    ret

_is_bigger:
    mov v10 [v25]


###########################################read int
_read_int:
    mov v1 0
_read_int_loop:
    ; first receive one character put it in al
    movb ah 0x08
    int 21h

    eq bl al 32
    #for space

    eq cl al 10
    #for enter

    orb bl bl cl
    jmp _read_int_end bl
    jmp is_digit 1
is_digit:
    subb bl al '0'
    IMUL v1 v1 10
    ADD v1 v1 bl
    jmp _read_int_loop 1
_read_int_end:
    RET

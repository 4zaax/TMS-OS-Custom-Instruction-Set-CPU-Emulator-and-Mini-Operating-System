_MAIN:
    showstr [0]     # show startup message
    showc '\n'
    _MAIN_LOOP:
    showstr [80]    # show >>> after each line
    AS idx 500      # user commands save from house 500
    call get_command
    call do_command
    JMP _MAIN_LOOP
END_MAIN:

get_command:
    push idx
    call get_line
    pop idx
    ret
END_get_command:


get_line:
    get c
    EQ eax c '\n'
    MOV [idx] '\0'
    RET eax
    MOV [idx] c
    INC idx
    JMP get_line
end_get_line:

save_char:
    MOV [idx] c
    INC idx
    RET
end_save_char:


do_command:
        call find_command
        JE command_error eax 0
        JE END eax 1
        JE CLS eax 2
        JE RUN eax 3
        command_error:
            showstr [60]
            showc '\n'
            jmp end_do_command
        cls:
            clear
            jmp end_do_command
        RUN:
            call run_program
            jmp end_do_command
end_do_command:
        RET


######################## turns eax 1 if command is equal to shut down ########################################################################
find_command:
    MOV ecx idx   # start index of user command
    MOV edx 90    # start index of shut down message command
    call str_cmp    # efx becomes 1 if ecx and edx are equal
    MUL eax efx 1
    RET eax
    MOV ecx idx     # start index of user command
    MOV edx 180     # start index of cls command
    call str_cmp    # efx becomes 1 if ecx and edx are equal
    MUL eax efx 2   # eax becomes 2 if efx is 1
    RET eax
    MOV ecx idx     # start index of user command
    MOV edx 190     # start index of run calculator command
    call str_cmp    # efx becomes 1 if ecx and edx are equal
    MUL eax efx 3   # eax becomes 3 if efx is 1
    RET eax
    RET
end_find_command:

############################# str_cmp checks ecx and edx turns efx 1 if 2 strings are equal and 0 if not ################################################
str_cmp:
    push ecx
    push edx
    AS efx 1
    str_cmp_loop:
        push ecx
        push edx
        mov ecx [ecx]
        mov edx [edx]
        eq efx ecx edx
        je finish efx 0 # characters were not equal so efx is 0
        eq efx ecx 0
        jmp finish efx  # characters were equal and both were 0 so efx should be 1
        ########################################################################################################################
        # now if none of those conditions are true it means that we have equal characters and string has not reached the end yet
        ########################################################################################################################
        pop edx
        pop ecx
        inc ecx
        inc edx
        jmp str_cmp_loop
        finish:
            pop edx
            pop ecx
        end_finish:
    end_str_cmp_loop:
    pop edx
    pop ecx
    ret
end_str_cmp:

load_program_data:
    push idx
    push ecs
    AS ecx idx
    add ecx idx 4
    AS edx [40]
    call str_cmp
    JE calc_exe efx 1
    JE calc_exe efx 0
    calc_exe:
        showstr [100]
        showc '\n'
        run ../calculator.txt 1
        showstr [140]
        jmp end_run_program
end_run_program:
    pop ecs
    pop idx
    RET

END:

_initial:
MOV cs 40000
MOV ds 50000
MOV ss 55000
MOV esp 60000

msg_result: db "the result value is :"  0
msg_welcome: db "please enter your numbers and operators like this <int> <operand> <int> "   10 0

################################################
MOV v20 hp
jmp start

start:
      mov esi msg_welcome
       movb ah 0x09
       INT 21h
   loop_prog:

       call get_line
       call parse
        mov esi msg_result
        movb ah 0x09
          INT 21h
       show v8
       jmp loop_prog

get_line:
  loop_line:
     call get_int
       mov [hp] v4
        add hp hp 4
       movb ah 0x08
       int 21h
       mov [hp] al
       eq bl al 10
       jmp return_get_line bl
      add hp hp 4
        movb ah 0x08
        int 21h
       eq bl al 101
       jmp end_program bl
       jmp loop_line

return_get_line:
 ret


get_int:
   mov v4 0
   int_loop:
     movb ah 0x08
     int 21h
     eq bl al 101
     jmp end_program bl
     eq bl al 32
     jmp end_input bl 
     sub v5 al '0'
     mul v4 v4 10
     add v4 v4 v5
     jmp int_loop

end_input:
  ret




parse:
   MOV v2 v20
   loop_parse2:
      eq bl [v2] 43
      call  ADD_overwrite  bl
      eq bl [v2] 45
      call  SUB_overwrite  bl
      eq bl [v2] 42
      call  MOL_overwrite  bl
      eq bl [v2] 47
      call  DIV_overwrite  bl
      eq bl [v2] 10
      ret bl
      add v2 v2 4
      jmp loop_parse2



ADD_overwrite:
  sub v2 v2 4
  MOV v8 [v2]
  add v2 v2 8
  MOV v9 [v2]
  call ADD_lable
  MOV [v2] v8
  ret

SUB_overwrite:
  sub v2 v2 4
  MOV v8 [v2]
  add v2 v2 8
  MOV v9 [v2]

  ret


MOL_overwrite:
  sub v2 v2 4
  MOV v8 [v2]
  add v2 v2 8
  MOV v9 [v2]
  call MUL_lable
  MOV [v2] v8
  ret

DIV_overwrite:
  sub v2 v2 4
  MOV v8 [v2]
  add v2 v2 8
  MOV v9 [v2]
  call DIV_lable
  MOV [v2] v8
  ret
